    <!-- JAVASCRIPT APP LOGIC -->
    <script>
        // Default State Structure
        const DEFAULT_STATE = {
            profiles: {
                userA: { name: 'พี่เซ้น', avatar: '👩‍❤️‍👨' },
                userB: { name: 'น้องแก้ม', avatar: '👩‍❤️‍💋‍👨' }
            },
            habits: [
                { id: 'h1', title: '🚿 อาบน้ำก่อน 19:30 น.', icon: '🚿' },
                { id: 'h2', title: '🛌 เข้านอนก่อน 23:00 น.', icon: '🛌' },
                { id: 'h3', title: '💧 ดื่มน้ำครบ 2 ลิตร', icon: '💧' }
            ],
            quotas: [
                { id: 'q1', title: '🧋 น้ำหวาน', icon: '🧋', max: 15, unit: 'แก้ว' },
                { id: 'q2', title: '🍦 ขนม / ของหวาน', icon: '🍦', max: 4, unit: 'ครั้ง' },
                { id: 'q3', title: '🥘 บุฟเฟต์ / ชาบู', icon: '🥘', max: 2, unit: 'มื้อ' },
                { id: 'q4', title: '🏃‍♀️ ออกกำลังกาย', icon: '🏃‍♀️', max: 10, unit: 'วัน' }
            ],
            dailyLogs: {}, // dateString -> { userA: { h1: true, note: '' }, userB: { ... } }
            quotaLogs: {}  // monthString -> { userA: { q1: 2 }, userB: { q1: 1 } }
        };

        let state = JSON.parse(localStorage.getItem('coupleTrackerState')) || DEFAULT_STATE;
        let currentUser = 'userA';
        let currentTab = 'daily';

        // Init App
        document.addEventListener('DOMContentLoaded', () => {
            // Set default date to today YYYY-MM-DD
            const today = new Date().toISOString().split('T')[0];
            document.getElementById('dailyDatePicker').value = today;

            // Set default month YYYY-MM
            const currentMonth = today.substring(0, 7);
            document.getElementById('quotaMonthPicker').value = currentMonth;

            updateProfileUI();
            loadDailyData();
            loadQuotaData();
            updateSummaryData();
        });

        function saveState() {
            localStorage.setItem('coupleTrackerState', JSON.stringify(state));
            updateSummaryData();
        }

        // --- PROFILE & USER SWITCHING ---
        function switchUser(userKey) {
            currentUser = userKey;
            updateProfileUI();
            loadDailyData();
            loadQuotaData();
        }

        function updateProfileUI() {
            const userA = state.profiles.userA;
            const userB = state.profiles.userB;

            document.getElementById('userANameDisplay').innerText = userA.name;
            document.getElementById('userAAvatar').innerText = userA.avatar;
            document.getElementById('userBNameDisplay').innerText = userB.name;
            document.getElementById('userBAvatar').innerText = userB.avatar;

            const btnA = document.getElementById('userATab');
            const btnB = document.getElementById('userBTab');

            if (currentUser === 'userA') {
                btnA.className = "py-2.5 px-3 rounded-xl font-medium text-sm flex items-center justify-center gap-2 transition duration-200 bg-white text-pink-600 shadow-md";
                btnB.className = "py-2.5 px-3 rounded-xl font-medium text-sm flex items-center justify-center gap-2 transition duration-200 text-white hover:bg-white/10";
            } else {
                btnB.className = "py-2.5 px-3 rounded-xl font-medium text-sm flex items-center justify-center gap-2 transition duration-200 bg-white text-indigo-600 shadow-md";
                btnA.className = "py-2.5 px-3 rounded-xl font-medium text-sm flex items-center justify-center gap-2 transition duration-200 text-white hover:bg-white/10";
            }
        }

        // --- TAB SWITCHING ---
        function switchMainTab(tabName) {
            currentTab = tabName;
            document.getElementById('tabDaily').classList.add('hidden');
            document.getElementById('tabQuota').classList.add('hidden');
            document.getElementById('tabSummary').classList.add('hidden');

            document.getElementById('tabDailyBtn').className = "flex-1 py-3 text-center border-b-2 border-transparent hover:text-slate-700 flex justify-center items-center gap-1.5 text-xs font-medium text-slate-500";
            document.getElementById('tabQuotaBtn').className = "flex-1 py-3 text-center border-b-2 border-transparent hover:text-slate-700 flex justify-center items-center gap-1.5 text-xs font-medium text-slate-500";
            document.getElementById('tabSummaryBtn').className = "flex-1 py-3 text-center border-b-2 border-transparent hover:text-slate-700 flex justify-center items-center gap-1.5 text-xs font-medium text-slate-500";

            if (tabName === 'daily') {
                document.getElementById('tabDaily').classList.remove('hidden');
                document.getElementById('tabDailyBtn').className = "flex-1 py-3 text-center border-b-2 border-pink-500 text-pink-600 flex justify-center items-center gap-1.5 text-xs font-semibold";
            } else if (tabName === 'quota') {
                document.getElementById('tabQuota').classList.remove('hidden');
                document.getElementById('tabQuotaBtn').className = "flex-1 py-3 text-center border-b-2 border-indigo-500 text-indigo-600 flex justify-center items-center gap-1.5 text-xs font-semibold";
            } else if (tabName === 'summary') {
                document.getElementById('tabSummary').classList.remove('hidden');
                document.getElementById('tabSummaryBtn').className = "flex-1 py-3 text-center border-b-2 border-amber-500 text-amber-600 flex justify-center items-center gap-1.5 text-xs font-semibold";
                updateSummaryData();
            }
        }

        // --- DAILY CHECKLIST LOGIC ---
        function loadDailyData() {
            const selectedDate = document.getElementById('dailyDatePicker').value;
            const habitsContainer = document.getElementById('habitsContainer');
            habitsContainer.innerHTML = '';

            const dayLog = (state.dailyLogs[selectedDate] && state.dailyLogs[selectedDate][currentUser]) || {};

            state.habits.forEach(habit => {
                const isChecked = !!dayLog[habit.id];
                const itemEl = document.createElement('div');
                itemEl.className = `p-3.5 rounded-2xl border transition duration-200 flex items-center justify-between ${isChecked ? 'bg-pink-50/80 border-pink-200' : 'bg-slate-50 border-slate-200'}`;

                itemEl.innerHTML = `
                    <div class="flex items-center gap-3">
                        <span class="text-xl">${habit.icon}</span>
                        <span class="text-xs font-medium text-slate-700">${habit.title}</span>
                    </div>
                    <label class="relative inline-flex items-center cursor-pointer">
                        <input type="checkbox" onchange="toggleHabit('${habit.id}', this.checked)" ${isChecked ? 'checked' : ''} class="sr-only peer">
                        <div class="w-11 h-6 bg-slate-200 peer-focus:outline-none rounded-full peer peer-checked:after:translate-x-full peer-checked:after:border-white after:content-[''] after:absolute after:top-[2px] after:left-[2px] after:bg-white after:border-slate-300 after:border after:rounded-full after:h-5 after:w-5 after:transition-all peer-checked:bg-pink-500"></div>
                    </label>
                `;
                habitsContainer.appendChild(itemEl);
            });

            // Note input
            document.getElementById('dailyNoteInput').value = dayLog.note || '';

            // Calculate simple streak count for this user
            calculateStreak();
        }

        function toggleHabit(habitId, checked) {
            const date = document.getElementById('dailyDatePicker').value;
            if (!state.dailyLogs[date]) state.dailyLogs[date] = {};
            if (!state.dailyLogs[date][currentUser]) state.dailyLogs[date][currentUser] = {};

            state.dailyLogs[date][currentUser][habitId] = checked;
            saveState();

            if (checked) {
                confetti({ particleCount: 30, spread: 60, origin: { y: 0.8 } });
            }
            loadDailyData();
        }

        function saveDailyNote() {
            const date = document.getElementById('dailyDatePicker').value;
            const noteText = document.getElementById('dailyNoteInput').value;

            if (!state.dailyLogs[date]) state.dailyLogs[date] = {};
            if (!state.dailyLogs[date][currentUser]) state.dailyLogs[date][currentUser] = {};

            state.dailyLogs[date][currentUser].note = noteText;
            saveState();
        }

        function toggleAddHabitForm() {
            document.getElementById('addHabitForm').classList.toggle('hidden');
        }

        function addNewHabit() {
            const title = document.getElementById('newHabitTitle').value.trim();
            const icon = document.getElementById('newHabitIcon').value.trim() || '✨';

            if (!title) return;

            const newId = 'h_' + Date.now();
            state.habits.push({ id: newId, title: `${icon} ${title}`, icon: icon });
            saveState();

            document.getElementById('newHabitTitle').value = '';
            toggleAddHabitForm();
            loadDailyData();
        }

        function calculateStreak() {
            let streak = 0;
            let checkDate = new Date();

            for (let i = 0; i < 30; i++) {
                const dateStr = checkDate.toISOString().split('T')[0];
                const dayLog = state.dailyLogs[dateStr] && state.dailyLogs[dateStr][currentUser];
                if (dayLog && Object.values(dayLog).some(v => v === true)) {
                    streak++;
                    checkDate.setDate(checkDate.getDate() - 1);
                } else if (i === 0) {
                    // Check yesterday if today hasn't been checked yet
                    checkDate.setDate(checkDate.getDate() - 1);
                } else {
                    break;
                }
            }
            document.getElementById('streakBadge').innerText = `🔥 ${streak} วันติด`;
        }

        // --- QUOTA TRACKER LOGIC ---
        function loadQuotaData() {
            const selectedMonth = document.getElementById('quotaMonthPicker').value;
            const container = document.getElementById('quotaCardsContainer');
            container.innerHTML = '';

            const monthLog = (state.quotaLogs[selectedMonth] && state.quotaLogs[selectedMonth][currentUser]) || {};

            state.quotas.forEach(quota => {
                const currentVal = monthLog[quota.id] || 0;
                const percent = Math.min(100, Math.round((currentVal / quota.max) * 100));

                let barColor = 'bg-indigo-500';
                if (percent >= 100) barColor = 'bg-red-500';
                else if (percent >= 75) barColor = 'bg-amber-500';

                const card = document.createElement('div');
                card.className = "bg-slate-50 p-3.5 rounded-2xl border border-slate-200 space-y-2.5";
                card.innerHTML = `
                    <div class="flex justify-between items-center">
                        <div class="flex items-center gap-2">
                            <span class="text-xl">${quota.icon}</span>
                            <div>
                                <h4 class="font-medium text-xs text-slate-800">${quota.title}</h4>
                                <p class="text-[11px] text-slate-400">เป้าหมายสูงสุด: ${quota.max} ${quota.unit}</p>
                            </div>
                        </div>
                        <!-- Stepper Buttons -->
                        <div class="flex items-center bg-white rounded-xl border border-slate-200 p-1 gap-2 shadow-sm">
                            <button onclick="updateQuotaVal('${quota.id}', -1)" class="w-6 h-6 rounded-lg bg-slate-100 hover:bg-slate-200 text-slate-600 font-bold text-xs flex items-center justify-center">-</button>
                            <span class="text-xs font-semibold text-slate-700 min-w-[24px] text-center">${currentVal}</span>
                            <button onclick="updateQuotaVal('${quota.id}', 1)" class="w-6 h-6 rounded-lg bg-indigo-50 hover:bg-indigo-100 text-indigo-600 font-bold text-xs flex items-center justify-center">+</button>
                        </div>
                    </div>

                    <!-- Progress Bar -->
                    <div class="space-y-1">
                        <div class="w-full bg-slate-200 rounded-full h-2 overflow-hidden">
                            <div class="${barColor} h-2 rounded-full transition-all duration-300" style="width: ${percent}%"></div>
                        </div>
                        <div class="flex justify-between text-[10px] text-slate-400 font-medium">
                            <span>ใช้ไปแล้ว ${currentVal} ${quota.unit}</span>
                            <span>${percent}%</span>
                        </div>
                    </div>
                `;
                container.appendChild(card);
            });
        }

        function updateQuotaVal(quotaId, delta) {
            const selectedMonth = document.getElementById('quotaMonthPicker').value;
            if (!state.quotaLogs[selectedMonth]) state.quotaLogs[selectedMonth] = {};
            if (!state.quotaLogs[selectedMonth][currentUser]) state.quotaLogs[selectedMonth][currentUser] = {};

            let currentVal = state.quotaLogs[selectedMonth][currentUser][quotaId] || 0;
            currentVal = Math.max(0, currentVal + delta);

            state.quotaLogs[selectedMonth][currentUser][quotaId] = currentVal;
            saveState();
            loadQuotaData();
        }

        function toggleAddQuotaForm() {
            document.getElementById('addQuotaForm').classList.toggle('hidden');
        }

        function addNewQuotaItem() {
            const title = document.getElementById('newQuotaTitle').value.trim();

            if (!title) return;

            const newId = 'q_' + Date.now();

            saveState();

            document.getElementById('newQuotaTitle').value = '';
            toggleAddQuotaForm();
            loadQuotaData();
        }

        // --- SUMMARY & LEADERBOARD LOGIC ---
        function updateSummaryData() {
            document.getElementById('userASummaryName').innerText = state.profiles.userA.name;
            document.getElementById('userASummaryAvatar').innerText = state.profiles.userA.avatar;
            document.getElementById('userBSummaryName').innerText = state.profiles.userB.name;
            document.getElementById('userBSummaryAvatar').innerText = state.profiles.userB.avatar;

            // Count total checked days for each user across all logged dates
            let scoreA = 0, scoreB = 0;
            Object.values(state.dailyLogs).forEach(day => {
                if (day.userA && Object.values(day.userA).some(v => v === true)) scoreA++;
                if (day.userB && Object.values(day.userB).some(v => v === true)) scoreB++;
            });

            document.getElementById('userADailyScore').innerText = scoreA;
            document.getElementById('userBDailyScore').innerText = scoreB;

            // Count total quotas used this month
            const currentMonth = document.getElementById('quotaMonthPicker').value || new Date().toISOString().substring(0, 7);
            let quotaCountA = 0, quotaCountB = 0;

            const monthData = state.quotaLogs[currentMonth];
            if (monthData) {
                if (monthData.userA) quotaCountA = Object.values(monthData.userA).reduce((a, b) => a + b, 0);
                if (monthData.userB) quotaCountB = Object.values(monthData.userB).reduce((a, b) => a + b, 0);
            }

            document.getElementById('userAQuotaUsed').innerText = quotaCountA;
            document.getElementById('userBQuotaUsed').innerText = quotaCountB;
        }
